#if UNITY_EDITOR
// The example-turns A/B's entry point (tools/live-runs/example-turns/README.md, #151). NOT a
// committed test: copy it into a detached copy of the repository as
// UnityProject/Packages/com.agenerela.framework/Tests/Runtime/ExampleTurnsRun.cs and run it in
// batch mode. It runs one model through EvalRunner, two arms interleaved, guards on:
//   block  the defaults: the few-shot examples as the system prompt's block
//   turns  DecideOptions.ExamplesAsTurns = true: the same examples as user/assistant turns
//
// Read from the environment:
//   AGENERELA_OLLAMA_MODEL  the model (LiveOllama.Model; qwen3.5:2b when unset)
//   AGENERELA_OLLAMA_URL    the server (LiveOllama.Endpoint; http://127.0.0.1:11434 when unset)
//   AGENERELA_OLLAMA_API_KEY the server's key, for one behind tools/ollama-server/'s gateway
//   AGENERELA_SWEEP_SET     the set file under Assets/ (Evaluation/Interim/agent.json when unset)
//   AGENERELA_SWEEP_PROFILE the profile asset (Assets/Evaluation/VillageGuard.asset when unset)
//   AGENERELA_EXAMPLE_STATEMENTS  1 gives every example and the idle example the statement in
//                           Statements below, on in-memory copies of the profile and its actions, in
//                           both arms (the second run). Unset, the profile's own, which has none
//
// Besides the report it writes tokens.json beside it: every answer's prompt and completion tokens
// and whether its examples went as turns, which the report does not carry.
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Threading.Tasks;
using Agenerela.Evaluation;
using Agenerela.Providers;
using Newtonsoft.Json;
using NUnit.Framework;
using UnityEditor;
using UnityEngine;
using Object = UnityEngine.Object;

namespace Agenerela.Tests
{
    public sealed class ExampleTurnsRun
    {
        private readonly List<Object> created = new List<Object>();

        [TearDown]
        public void TearDown()
        {
            foreach (var obj in created)
            {
                Object.DestroyImmediate(obj);
            }

            created.Clear();
        }

        [Test]
        [Category(LiveOllama.Category)]
        [Timeout(14400000)] // 4 h. The Test Framework's default, 180 s, ends a 389-prompt run halfway.
        public async Task RunOneModel()
        {
            LiveOllama.Reset();
            LiveOllama.RequireReachable();

            string setRelative = Environment.GetEnvironmentVariable("AGENERELA_SWEEP_SET") ?? "Evaluation/Interim/agent.json";
            string profilePath = Environment.GetEnvironmentVariable("AGENERELA_SWEEP_PROFILE") ?? "Assets/Evaluation/VillageGuard.asset";
            var set = PromptSet.Load(Path.GetFullPath(Path.Combine(Application.dataPath, setRelative)));
            var profile = AssetDatabase.LoadAssetAtPath<AgentProfile>(profilePath);
            Assert.That(profile, Is.Not.Null, $"No AgentProfile at {profilePath}");
            if (Environment.GetEnvironmentVariable("AGENERELA_EXAMPLE_STATEMENTS") == "1")
            {
                profile = WithStatements(profile, set);
            }

            var block = new EvalArm("block", Config("turns-A"), new DecideOptions());
            var turns = new EvalArm("turns", Config("turns-B"), new DecideOptions { ExamplesAsTurns = true });
            var runner = new EvalRunner(set, profile, block, turns)
            {
                Order = EvalArmOrder.Interleaved,
                // The wording the 9 November 2026 sweep used, so results compare.
                DescribeState = state => new[] { (bool)state["isFollowing"] ? "You are following the player." : "You are not following the player." },
            };

            var started = DateTime.Now;
            var report = await LiveOllama.NotCancelled(runner.RunAsync());
            var took = DateTime.Now - started;
            string saved = report.Save(Path.Combine(Application.dataPath, "..", "Logs", "Evaluation"));

            var tokens = report.Answers.Select(a => new
            {
                arm = a.Arm,
                promptId = a.PromptId,
                promptTokens = a.Result?.Telemetry.PromptTokens ?? 0,
                completionTokens = a.Result?.Telemetry.CompletionTokens ?? 0,
                examplesSentAsTurns = a.Result?.Telemetry.ExamplesSentAsTurns ?? false,
                exampleTurns = a.Result?.Request?.ExampleTurns?.Count ?? 0,
                hasBlock = !string.IsNullOrEmpty(a.Result?.Request?.FewShotBlock),
            }).ToList();
            string folder = Directory.Exists(saved) ? saved : Path.GetDirectoryName(saved);
            File.WriteAllText(Path.Combine(folder, "tokens.json"), JsonConvert.SerializeObject(tokens, Formatting.Indented));

            string line = $"EVAL-REPORT {saved}\nEVAL-TOOK {took.TotalSeconds:F0}s for {set.Prompts.Count()} prompts per arm";
            TestContext.Out.WriteLine(line);
            Debug.Log(line);
        }

        // The statements of the second run (9 November 2026), one short in-character line per
        // VillageGuard example and its idle example. Written on copies made in memory, so the asset
        // and everything else that reads it stay as they are. Keyed by action id; {0} is the
        // example's target, as in its stimulus.
        private static readonly Dictionary<string, string> Statements = new Dictionary<string, string>
        {
            ["attack_target"] = "Steel out. The {0} won't trouble us long.",
            ["follow_player"] = "Right behind you.",
            ["move_to"] = "Heading to the {0} now.",
            ["stop_following"] = "I'll hold here, then.",
        };

        private const string IdleStatement = "Aye, quiet so far.";

        private AgentProfile WithStatements(AgentProfile original, PromptSet set)
        {
            var copy = Object.Instantiate(original);
            copy.name = original.name;
            created.Add(copy);
            copy.Actions = original.Actions.Select(asset =>
            {
                var action = Object.Instantiate(asset);
                action.name = asset.name;
                created.Add(action);
                Assert.That(Statements.TryGetValue(action.Action.Id, out string line), Is.True, $"No statement written for {action.Action.Id}");
                action.Action.ExampleStatement = line;
                return action;
            }).ToList();
            copy.IdleExampleStatement = IdleStatement;
            Assert.That(original.Actions.All(a => string.IsNullOrEmpty(a.Action.ExampleStatement)), Is.True, "The asset itself is unchanged");

            // Hard rule 4: no statement may be an evaluation prompt.
            int before = PromptSetChecks.Check(set, original).Count(p => p.Rule == PromptRule.StimulusIsExample);
            var after = PromptSetChecks.Check(set, copy).Where(p => p.Rule == PromptRule.StimulusIsExample).ToList();
            Assert.That(after.Count, Is.EqualTo(before), "A statement matches a prompt: " + string.Join("; ", after));

            string line2 = "EXAMPLE-STATEMENTS " + string.Join(" | ", copy.Actions.Select(a => $"{a.Action.Id}: {a.Action.ExampleStatement}")) + $" | idle: {copy.IdleExampleStatement}";
            TestContext.Out.WriteLine(line2);
            Debug.Log(line2);
            return copy;
        }

        private ProviderConfig Config(string name)
        {
            var config = ScriptableObject.CreateInstance<OllamaProviderConfig>();
            config.name = name;
            config.Endpoint = LiveOllama.Endpoint;
            config.Model = LiveOllama.Model;
            if (LiveOllama.ApiKey.Length > 0)
            {
                config.ApiKeyVariable = LiveOllama.ApiKeyVariable;
            }

            created.Add(config);
            return config;
        }
    }
}
#endif
