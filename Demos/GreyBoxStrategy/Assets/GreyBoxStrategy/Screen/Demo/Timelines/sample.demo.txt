# Painted Realm, the demo video's sample: two moments of the framework at work, about 55 s.
# The game is a live one: five qwen3.5:4b rulers with personas, seed 123, played on 12 November and
# replayed here, so no model is called and every take is the same. Steps: DemoScript.cs.
# Turn 4 has both moments: Valen takes Stonereach, and Merrow's answer for the same province is caught
# by the target grounding guard.

load Assets/GreyBoxStrategy/Screen/Replays/seed-0123-live-4b.json
# Larger than the game's own 1080p size, so the screen reads in a video; 1.2 keeps the full layout.
ui-scale 1.2
turn 4
until Taxes kesh
roll
fade in time=0.7
title "Painted Realm" "Five rulers, each a qwen3.5:4b model behind an Agenerela agent. A recorded game, replayed." time=3.6

# ---- 1: a ruler's turn ----
chapter "THE FRAMEWORK AT WORK  ·  1 OF 2" "A ruler's turn"
speed 2
until Capture valen
speed 1
after Capture valen
wait 0.4
click ".chronicle .lg'Valen takes Stonereach'"
wait 0.25
click ".chronicle .lg.open .lg-link"
wait 0.5
callout ".inspector .code-box" "The model's reply, field by field" "Action first, then target, then the statement: the order measured to work best." side=left
wait 3.6
clear
click ".inspector .ktab:has('Offered')"
wait 0.3
callout ".inspector .chips" "Only registered, legal actions" "The game registers its moves; the rules say which Valen may take this turn. Nothing else is in the schema." side=left
wait 3.8
clear
click ".inspector .ktab:has('Guards')"
wait 0.3
callout ".inspector .sec:nth(1)" "Every answer is checked" "The schema, then the target's grounding against a 60% bar, then the rules once more when the move lands." side=left
wait 3.8
clear
click ".inspector .panel-head .tbtn"
map fit time=0.9

# ---- 2: a guard catches a bad answer ----
chapter "THE FRAMEWORK AT WORK  ·  2 OF 2" "A guard catches a bad answer"
after Hold merrow
wait 0.3
click ".chronicle .lg'guard stopped'"
wait 0.25
click ".chronicle .lg.open .lg-link"
wait 0.3
# The inspector opens on the tab last shown.
click ".inspector .ktab:has('Answer')"
wait 0.3
callout ".inspector .code-box" "Merrow wanted Stonereach too" "Its reply named the province Valen was taking. The agent ran no action instead." side=left
wait 3.4
clear
click ".inspector .ktab:has('Guards')"
wait 0.3
dim ".inspector .sec:nth(1)" pad=8
callout ".inspector .gline:has('TargetGroundingGuard')" "A guard stopped it" "The target grounding guard: under 1% of the model's own probability was on a place it may name. The agent ran no action, and the game was never asked." side=left
wait 4.6
clear
click ".inspector .panel-head .tbtn"
map fit time=0.9
hide-cursor
clear-chapter
wait 0.4
title "The model chooses. The framework checks." "The game runs only moves it registered, after every guard has passed them." time=3.4
fade out time=0.8
