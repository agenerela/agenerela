using System;
using System.Collections.Generic;
using System.Globalization;
using System.IO;
using System.Linq;
using System.Threading.Tasks;
using Agenerela.Evaluation;
using Agenerela.Providers;
using NUnit.Framework;
using UnityEditor;
using UnityEngine;
using Object = UnityEngine.Object;

namespace Agenerela.Tests
{
    public sealed class Phase4Run
    {
        private readonly List<Object> created = new List<Object>();

        [TearDown]
        public void TearDown()
        {
            foreach (var obj in created)
            {
                Object.DestroyImmediate(obj);
            }

            created.Clear();
        }

        [Test]
        [Category(LiveOllama.Category)]
        [Timeout(14400000)]
        public async Task RunOne()
        {
            LiveOllama.Reset();
            LiveOllama.RequireReachable();

            string run = Environment.GetEnvironmentVariable("AGENERELA_P4_RUN") ?? "";
            // Request 007: AGENERELA_P4_SET points at another set file; unset is the 209 set.
            string setPath = Environment.GetEnvironmentVariable("AGENERELA_P4_SET");
            var set = PromptSet.Load(Path.GetFullPath(string.IsNullOrEmpty(setPath)
                ? Path.Combine(Application.dataPath, "Evaluation/agent.json")
                : setPath));
            var profile = AssetDatabase.LoadAssetAtPath<AgentProfile>("Assets/Evaluation/VillageGuard.asset");
            Assert.That(profile, Is.Not.Null);
            // T1 (think-ab, 24 November): a slice of the set, every 7th prompt within each category, in memory only.
            if (int.TryParse(Environment.GetEnvironmentVariable("AGENERELA_P4_SLICE"), out int k))
            {
                foreach (var c in set.Categories)
                {
                    var keep = c.Prompts.Where((p, i) => i % 7 == k).ToList();
                    c.Prompts.Clear();
                    c.Prompts.AddRange(keep);
                }

                TestContext.Out.WriteLine($"SLICE {k}: {set.Prompts.Count()} prompts");
                Debug.Log($"SLICE {k}: {set.Prompts.Count()} prompts");
            }

            EvalArm a, b;
            var order = EvalArmOrder.Interleaved;
            string model = LiveOllama.Model;
            switch (run)
            {
                case "gate":
                    // Guards on (the defaults) against CheckTarget = false; the runner adds no developer guards.
                    a = new EvalArm("guards-on", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("guards-off", Config("p4-B", model), new DecideOptions { CheckTarget = false });
                    break;
                case "idle":
                    a = new EvalArm("idle-profile", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("idle-none", Config("p4-B", model), new DecideOptions { IdleExampleStimulus = "" });
                    break;
                case "list":
                    a = new EvalArm("list-off", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("list-on", Config("p4-B", model), new DecideOptions { ListActionsAndTargets = true });
                    break;
                case "grounding":
                    float low = float.Parse(Environment.GetEnvironmentVariable("AGENERELA_P4_LOW") ?? "0.6", CultureInfo.InvariantCulture);
                    a = new EvalArm("ground-0.8", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("ground-" + low.ToString(CultureInfo.InvariantCulture), Config("p4-B", model),
                        new DecideOptions { TargetGroundingThreshold = low });
                    break;
                case "models":
                    a = new EvalArm("qwen3.5-2b", Config("p4-A", "qwen3.5:2b"), new DecideOptions());
                    b = new EvalArm("qwen3.5-4b", Config("p4-B", "qwen3.5:4b"), new DecideOptions());
                    order = EvalArmOrder.Sequential;
                    break;
                case "rename":
                    // DR-008: the free-text field as `statement` (the default) against the prototype's `dialogue`.
                    a = new EvalArm("statement", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("dialogue", Config("p4-B", model), new DecideOptions { StatementFieldName = DecisionSchema.DialogueFieldName });
                    break;
                case "targetrule":
                    // #107: probabilities and the grounding guard (the default) against no probabilities, which sends the one-line target rule.
                    a = new EvalArm("probabilities", Config("p4-A", model), new DecideOptions());
                    var noProbs = (OllamaProviderConfig)Config("p4-B", model);
                    noProbs.TokenProbabilities = false;
                    b = new EvalArm("target-rule", noProbs, new DecideOptions());
                    break;
                case "history":
                    a = new EvalArm("history", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("no-history", Config("p4-B", model), new DecideOptions()) { IgnoreHistory = true };
                    break;
                case "freechoice":
                    // §2.5: one added example of a free choice (wording approved by the tech lead, 23 November).
                    a = new EvalArm("plain", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("free-choice", Config("p4-B", model), new DecideOptions
                    {
                        AddedExamples = new List<FewShotExample> { new FewShotExample("Go wherever you think you're needed.", "move_to", "north_tower", "To the tower, then.") },
                    });
                    break;
                // The four runs #384 made possible (live-runs.md, "Ready to run now"); test after 4485e83.
                case "summarise":
                    a = new EvalArm("templated", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("summarised", Config("p4-B", model), new DecideOptions()) { SummariseTargets = true };
                    break;
                case "pair":
                    a = new EvalArm("flat", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("paired", Config("p4-B", model), new DecideOptions { PairTargetsWithActions = true });
                    break;
                case "statements":
                    a = new EvalArm("no-statements", Config("p4-A", model), new DecideOptions { ExampleStatements = false });
                    b = new EvalArm("statements", Config("p4-B", model), new DecideOptions());
                    break;
                case "counters":
                    a = new EvalArm("no-counters", Config("p4-A", model), new DecideOptions { CounterExamples = false });
                    b = new EvalArm("counters", Config("p4-B", model), new DecideOptions());
                    break;
                case "think":
                    // DR-029, T1: both arms at num_ctx 16384 and 300 s; think is the only difference.
                    a = new EvalArm("think-off", Thinking(Config("p4-A", model), false), new DecideOptions());
                    b = new EvalArm("think-on", Thinking(Config("p4-B", model), true), new DecideOptions());
                    break;
                case "claims":
                    // DR-031: the claim rule. Only the switch differs between the arms.
                    a = new EvalArm("claims-off", Config("p4-A", model), new DecideOptions());
                    b = new EvalArm("claims-on", Config("p4-B", model), new DecideOptions { StimulusIsClaim = true });
                    break;
                default:
                    Assert.Fail($"AGENERELA_P4_RUN '{run}' is not one of gate, idle, list, grounding, models, rename, targetrule, history, freechoice, summarise, pair, statements, counters, think, claims");
                    return;
            }

            var runner = new EvalRunner(set, profile, a, b)
            {
                Order = order,
                // The wording every evaluation-set run since 9 November 2026 used.
                WarmUpRequests = run == "think" ? 0 : 3,
                DescribeState = state => new[] { (bool)state["isFollowing"] ? "You are following the player." : "You are not following the player." },
            };

            var started = DateTime.Now;
            var report = await LiveOllama.NotCancelled(runner.RunAsync());
            var took = DateTime.Now - started;
            string saved = report.Save(Path.Combine(Application.dataPath, "..", "Logs", "Evaluation"));
            // Per decision: the provider's own token counts (prompt_eval_count, eval_count) and latency, which
            // report.json leaves out (24 November, the ListActionsAndTargets timing run).
            var csv = new System.Text.StringBuilder("arm,promptId,category,outcome,latencySeconds,promptTokens,completionTokens\n");
            foreach (var ans in report.Answers)
            {
                var t = ans.Result?.Telemetry;
                csv.Append(ans.Arm).Append(',').Append(ans.PromptId).Append(',').Append(ans.Category).Append(',').Append(ans.Outcome).Append(',')
                   .Append(ans.LatencySeconds.ToString("F4", CultureInfo.InvariantCulture)).Append(',')
                   .Append(t?.PromptTokens ?? -1).Append(',').Append(t?.CompletionTokens ?? -1).Append('\n');
            }

            File.WriteAllText(Path.Combine(saved, "decisions.csv"), csv.ToString());
            string line = $"EVAL-REPORT {saved}\nEVAL-TOOK {took.TotalSeconds:F0}s for {set.Prompts.Count()} prompts per arm";
            TestContext.Out.WriteLine(line);
            Debug.Log(line);
        }

        // DR-029: the model's own thinking channel, not a reasoning field in the schema.
        private static ProviderConfig Thinking(ProviderConfig config, bool think)
        {
            var c = (OllamaProviderConfig)config;
            c.Think = think;
            c.ContextSize = 16384;
            c.TimeoutSeconds = 300f;
            return c;
        }

        private ProviderConfig Config(string name, string model)
        {
            var config = ScriptableObject.CreateInstance<OllamaProviderConfig>();
            config.name = name;
            config.Endpoint = LiveOllama.Endpoint;
            config.Model = model;
            if (LiveOllama.ApiKey.Length > 0)
            {
                config.ApiKeyVariable = LiveOllama.ApiKeyVariable;
            }

            created.Add(config);
            return config;
        }
    }
}
